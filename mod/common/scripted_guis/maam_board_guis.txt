# Card click: select scope:regiment for the bench.
maam_board_select = {
	scope = character
	is_valid = {
		maam_board_actions_allowed = yes
		exists = scope:regiment
		OR = {
			scope:regiment = { is_personal_maa_regiment = yes regiment_owner = root }
			scope:regiment = { maa_manager_title_regiment_owned = yes }
		}
	}
	effect = {
		if = {
			limit = {
				maam_board_actions_allowed = yes
				exists = scope:regiment
			}
			maam_board_select_effect = yes
		}
	}
}

# Kin card click: pair scope:target_regiment; GUI passes scope:type_relation.
maam_board_pair = {
	scope = character
	is_valid = {
		NOT = { has_variable = maam_rt_active }
		maa_manager_select_target_valid = yes
	}
	effect = {
		if = {
			limit = { NOT = { has_variable = maam_rt_active } }
			maam_board_set_partner_effect = yes
		}
	}
}

######################################################
# Card display helpers (scope:regiment = the card's regiment)
######################################################

# A selection exists on the same board as scope:regiment (personal vs title).
maam_board_has_selection = {
	scope = character
	is_shown = {
		has_variable = maam_board_selected
		exists = var:maa_manager_selected_regiment_scope
		exists = scope:regiment
		OR = {
			AND = {
				scope:regiment = { is_title_maa_regiment = yes }
				var:maa_manager_selected_regiment_scope = { is_title_maa_regiment = yes }
			}
			AND = {
				scope:regiment = { is_title_maa_regiment = no }
				var:maa_manager_selected_regiment_scope = { is_title_maa_regiment = no }
			}
		}
	}
}

maam_board_card_selected = {
	scope = character
	is_shown = {
		has_variable = maam_board_selected
		exists = var:maa_manager_selected_regiment_scope
		exists = scope:regiment
		scope:regiment = var:maa_manager_selected_regiment_scope
	}
}

maam_board_card_partner = {
	scope = character
	is_shown = {
		has_variable = maam_board_selected
		exists = var:maa_manager_selected_target_scope
		exists = scope:regiment
		scope:regiment = var:maa_manager_selected_target_scope
	}
}

# Could be paired with the selection (type equality is added by the GUI).
maam_board_card_kin = {
	scope = character
	is_shown = {
		NOT = { has_variable = maam_rt_active }
		has_variable = maam_board_selected
		exists = var:maa_manager_selected_regiment_scope
		maa_manager_can_modify_regiments = yes
		exists = scope:regiment
		NOT = { scope:regiment = var:maa_manager_selected_regiment_scope }
		trigger_if = {
			limit = { exists = var:maa_manager_selected_target_scope }
			NOT = { scope:regiment = var:maa_manager_selected_target_scope }
		}
		scope:regiment = { maa_manager_compatible_target_base = yes }
	}
}

}

# Title regiments and unsupported rulers: the existing retrain list.
maam_open_legacy_retrain = {
	scope = character
	is_valid = {
		maa_manager_can_modify_regiments = yes
		exists = var:maa_manager_selected_regiment_scope
		var:maa_manager_selected_regiment_scope = { maa_manager_is_manageable_regiment = yes }
		NOT = { has_variable = maam_rt_active }
	}
	effect = {
		if = {
			limit = { exists = var:maa_manager_selected_regiment_scope }
			set_variable = { name = maam_preselect_regiment value = var:maa_manager_selected_regiment_scope }
		}
	}
}

# Tile click goes to selection (not the partner, nothing while retraining).
maam_board_card_plain = {
	scope = character
	is_shown = {
		NOT = { has_variable = maam_rt_active }
		exists = scope:regiment
		trigger_if = {
			limit = { has_variable = maam_board_selected exists = var:maa_manager_selected_regiment_scope }
			NOT = { scope:regiment = var:maa_manager_selected_regiment_scope }
		}
		trigger_if = {
			limit = { exists = var:maa_manager_selected_target_scope }
			NOT = { scope:regiment = var:maa_manager_selected_target_scope }
		}
	}
}

# A role banner replaces the type name on the tile.
maam_board_card_has_banner = {
	scope = character
	is_shown = {
		exists = scope:regiment
		OR = {
			AND = {
				has_variable = maam_board_selected
				exists = var:maa_manager_selected_regiment_scope
				scope:regiment = var:maa_manager_selected_regiment_scope
			}
			AND = {
				has_variable = maam_board_selected
				exists = var:maa_manager_selected_target_scope
				scope:regiment = var:maa_manager_selected_target_scope
			}
			AND = {
				has_variable = maam_rt_active
				exists = var:maam_rt_source
				scope:regiment = var:maam_rt_source
			}
			AND = {
				has_variable = maam_rt_active
				exists = var:maam_rt_candidate
				scope:regiment = var:maam_rt_candidate
			}
		}
	}
}

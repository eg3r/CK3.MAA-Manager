maam_ob_lift = {
	scope = character
	is_valid = {
		exists = scope:regiment
		OR = {
			AND = {
				maam_ob_kind_is = { KIND = lift }
				exists = var:maam_ob_src
				var:maam_ob_src = scope:regiment
			}
			AND = {
				maam_ob_can_order = yes
				NOT = { exists = var:maam_ob_kind }
				scope:regiment = {
					maam_ob_regiment_ok = yes
					maa_size > 1
				}
			}
		}
	}
	effect = { maam_ob_lift_effect = yes }
}

}

# scope:want = flag:lift / flag:ended / flag:merge / flag:pour
maam_ob_kind_gui = {
	scope = character
	is_shown = {
		exists = var:maam_ob_kind
		exists = scope:want
		var:maam_ob_kind = scope:want
		maam_ob_src_ok = yes
	}
}

# Any order is open (view switches, folding and settings wait).
maam_ob_has_order = {
	scope = character
	is_shown = {
		exists = var:maam_ob_kind
		maam_ob_src_ok = yes
	}
}

# No order and nothing blocks: lanes are live.
maam_ob_idle = {
	scope = character
	is_shown = {
		maam_ob_can_order = yes
		NOT = {
			AND = {
				exists = var:maam_ob_kind
				maam_ob_src_ok = yes
			}
		}
	}
}

# The lane's regiment can take part in orders (title loans are read-only).
maam_ob_lane_ok = {
	scope = character
	is_shown = {
		exists = scope:regiment
		scope:regiment = { maam_ob_regiment_ok = yes }
	}
}

maam_ob_is_src = {
	scope = character
	is_shown = {
		exists = var:maam_ob_kind
		exists = var:maam_ob_src
		exists = scope:regiment
		scope:regiment = var:maam_ob_src
	}
}

# Lane can receive a lifted stack (the GUI adds the type equality).
maam_ob_lane_receive = {
	scope = character
	is_shown = {
		maam_ob_can_order = yes
		maam_ob_kind_is = { KIND = lift }
		maam_ob_src_ok = yes
		var:maam_ob_src = { maa_size > 1 }
		exists = scope:regiment
		scope:regiment = {
			maam_ob_regiment_ok = yes
			NOT = { this = root.var:maam_ob_src }
			maam_ob_same_board = { OTHER = root.var:maam_ob_src }
			can_upgrade_maa = yes
		}
	}
}

# Lane can take the source's last stack (merge or lift ended).
maam_ob_lane_merge = {
	scope = character
	is_shown = {
		maam_ob_can_order = yes
		OR = {
			maam_ob_kind_is = { KIND = merge }
			maam_ob_kind_is = { KIND = ended }
		}
		maam_ob_src_ok = yes
		var:maam_ob_src = { maa_size = 1 }
		exists = scope:regiment
		scope:regiment = {
			maam_ob_regiment_ok = yes
			NOT = { this = root.var:maam_ob_src }
			maam_ob_same_board = { OTHER = root.var:maam_ob_src }
			can_upgrade_maa = yes
		}
	}
}

# Quartermaster: scope:want = flag:grow / flag:combine (both regiments still valid).
maam_qm_is = {
	scope = character
	is_shown = {
		exists = var:maam_qm_kind
		exists = scope:want
		var:maam_qm_kind = scope:want
		maam_qm_tip_valid = yes
	}
}

# A tip is stored but no longer holds: the strip asks again by itself.
maam_qm_stale = {
	scope = character
	is_shown = {
		exists = var:maam_qm_kind
		NOT = { maam_qm_tip_valid = yes }
	}
}

# Show me: the lane is one of the highlighted pair.
maam_qm_hl_is = {
	scope = character
	is_shown = {
		exists = var:maam_qm_hl
		exists = scope:regiment
		OR = {
			AND = {
				exists = var:maam_qm_a
				scope:regiment = var:maam_qm_a
			}
			AND = {
				exists = var:maam_qm_b
				scope:regiment = var:maam_qm_b
			}
		}
	}
}

# Default view (unset counts as the list).
maam_pref_is = {
	scope = character
	is_shown = {
		exists = scope:want
		trigger_if = {
			limit = { exists = var:maam_pref_view }
			var:maam_pref_view = scope:want
		}
		trigger_else = { scope:want = flag:list }
	}
}

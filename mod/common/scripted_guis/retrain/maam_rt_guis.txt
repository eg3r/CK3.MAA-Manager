######################################################
# MAA Manager - Dynamic retrain scripted GUIs
#
# Every action is an explicit click (or a trigger_when state on a visible
# widget for capture/quote). Window hiding never executes anything here.
# Type equality is GUI ObjectsEqual, passed as scope:real_relation together
# with the exact regiment scopes; the backend revalidates everything else.
######################################################

# Start for scope:real_source: adds the temporary slot and arms the quote.
maam_rt_start = {
	scope = character
	is_valid = { maam_rt_start_allowed = yes }
	effect = {
		if = {
			limit = { maam_rt_start_allowed = yes }
			maam_rt_start_effect = yes
		}
	}
}

# Snapshot right before the native purchase (same click as MenAtArmsTypeView.Create).
maam_rt_begin = {
	scope = character
	is_valid = { maam_rt_can_buy = yes }
	effect = {
		if = {
			limit = {
				maam_rt_can_buy = yes
				exists = scope:real_source
				exists = scope:real_relation
				scope:real_source = var:maam_rt_source
			}
			maam_rt_begin_effect = yes
		}
	}
}

# Capture the purchase (button, and automatic once the new regiment exists).
maam_rt_capture = {
	scope = character
	is_valid = { maam_rt_capture_ready = yes }
	effect = {
		if = {
			limit = { maam_rt_capture_ready = yes }
			maam_rt_capture_effect = yes
		}
	}
}

# Quote phase: record and refund. Target phase: commit (grow, settle, replace).
maam_rt_accept = {
	scope = character
	is_valid = {
		maam_rt_purchase_intact = yes
		has_variable = maam_rt_captured
	}
	effect = { maam_rt_accept_effect = yes }
}

maam_rt_rollback = {
	scope = character
	is_valid = { maam_rt_can_rollback = yes }
	effect = { maam_rt_rollback_effect = yes }
}

maam_rt_release = {
	scope = character
	is_valid = { maam_rt_can_release = yes }
	effect = { maam_rt_release_effect = yes }
}

# Display: a purchase exists that could not be verified.
maam_rt_capture_failed_gui = {
	scope = character
	is_shown = { maam_rt_can_release = yes }
}

maam_rt_cancel = {
	scope = character
	is_valid = { maam_rt_can_cancel = yes }
	effect = { maam_rt_cancel_effect = yes }
}

maam_rt_recover = {
	scope = character
	is_valid = { maam_rt_can_recover_expired = yes }
	effect = { maam_rt_recover_effect = yes }
}

maam_rt_abandon = {
	scope = character
	is_valid = {
		has_variable = maam_rt_active
		NOT = { has_variable = maam_rt_click_guard }
		NOT = { maam_rt_can_recover_expired = yes }
		NOT = { has_variable = maam_rt_settled }
	}
	effect = { maam_rt_abandon_effect = yes }
}

maam_rt_unpend = {
	scope = character
	is_valid = {
		has_variable = maam_rt_active
		has_variable = maam_rt_pending
		has_variable = maam_rt_count_before
		maa_regiments_count = var:maam_rt_count_before
		gold = var:maam_rt_gold_before
	}
	effect = { maam_rt_unpend_effect = yes }
}

maam_rt_dismiss_result = {
	scope = character
	effect = { maam_rt_clear_result = yes }
}

######################################################
# Display helpers (is_shown only)
######################################################

# Ruler can use the measured retrain at all.
maam_rt_ruler_supported = {
	scope = character
	is_shown = { maam_rt_supported_ruler = yes }
}

# Phase: choose and buy the test stack of the current type.
maam_rt_phase_quote = {
	scope = character
	is_shown = {
		has_variable = maam_rt_active
		has_variable = maam_rt_need_source
		has_variable = maam_rt_armed
	}
}

# Phase: choose and buy the target stack.
maam_rt_phase_target = {
	scope = character
	is_shown = {
		has_variable = maam_rt_active
		NOT = { has_variable = maam_rt_need_source }
		has_variable = maam_rt_armed
	}
}

# A native purchase was sent and is not captured yet.
maam_rt_phase_pending = {
	scope = character
	is_shown = {
		has_variable = maam_rt_active
		has_variable = maam_rt_pending
	}
}

# Test stack captured: record price and refund.
maam_rt_phase_record = {
	scope = character
	is_shown = {
		has_variable = maam_rt_active
		has_variable = maam_rt_captured
		has_variable = maam_rt_need_source
	}
}

# Target stack captured: receipt.
maam_rt_phase_receipt = {
	scope = character
	is_shown = {
		has_variable = maam_rt_active
		has_variable = maam_rt_captured
		NOT = { has_variable = maam_rt_need_source }
	}
}

maam_rt_interrupted_gui = {
	scope = character
	is_shown = { maam_rt_interrupted = yes }
}

maam_rt_tile_is_source = {
	scope = character
	is_shown = {
		has_variable = maam_rt_active
		exists = var:maam_rt_source
		exists = scope:regiment
		scope:regiment = var:maam_rt_source
	}
}

maam_rt_tile_is_test = {
	scope = character
	is_shown = {
		has_variable = maam_rt_active
		exists = var:maam_rt_candidate
		exists = scope:regiment
		scope:regiment = var:maam_rt_candidate
	}
}

# Display: which result to show (scope:want is a flag).
maam_rt_result_is = {
	scope = character
	is_shown = {
		exists = var:maam_rt_result
		exists = scope:want
		var:maam_rt_result = scope:want
	}
}

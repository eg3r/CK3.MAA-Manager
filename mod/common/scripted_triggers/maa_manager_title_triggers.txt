# Title mode is stored on the player, not only in the client VariableSystem.
# All regiment triggers below run with ROOT = the managing character.
maa_manager_is_title_mode = {
	has_variable = maa_manager_title_mode
}

maa_manager_title_regiment_owned = {
	is_title_maa_regiment = yes
	exists = regiment_owning_title
	regiment_owning_title = { holder = root }
}

maa_manager_title_regiment_manageable = {
	maa_manager_title_regiment_owned = yes
	is_raised = no
	exists = regiment_controlling_title
	regiment_controlling_title = regiment_owning_title
	regiment_controller = root
}

# Parameter is a literal database key supplied by existing per-type triggers.
# The native title gate includes restrictions not present in personal triggers.
maa_manager_can_recruit_title_type = {
	exists = var:maa_manager_selected_regiment_scope
	var:maa_manager_selected_regiment_scope = {
		maa_manager_title_regiment_manageable = yes
		NOT = { is_maa_type = $TYPE$ }
		regiment_owning_title = { can_create_title_maa = $TYPE$ }
	}
}

maa_manager_title_retrain_source_valid = {
	exists = var:maa_manager_selected_regiment_scope
	var:maa_manager_selected_regiment_scope = {
		maa_manager_title_regiment_manageable = yes
		# No exported effect restores a station after recreation.
		NOT = { exists = regiment_station }
	}
}

maa_manager_title_uses_treasury = {
	maa_manager_is_title_mode = yes
	has_treasury = yes
	government_allows = replace_gold_cost_by_treasury
}

# Presentation and payment use the same rule; retain personal-mode behavior.
maa_manager_retrain_uses_treasury = {
	trigger_if = {
		limit = { maa_manager_is_title_mode = yes }
		maa_manager_title_uses_treasury = yes
	}
	trigger_else = { has_treasury = yes }
}

# A candidate target must be manageable now, not just in a cached list.
maa_manager_compatible_target = {
	maa_manager_is_manageable_regiment = yes
	NOT = { this = root.var:maa_manager_selected_regiment_scope }
	maa_manager_matches_source_type = yes
	trigger_if = {
		limit = { root = { maa_manager_is_title_mode = yes } }
		regiment_owning_title = root.var:maa_manager_selected_regiment_scope.regiment_owning_title
	}
}

maa_manager_select_source_valid = {
	maa_manager_can_modify_regiments = yes
	exists = scope:regiment
	scope:regiment = { maa_manager_is_manageable_regiment = yes }
}

maa_manager_select_target_valid = {
	maa_manager_can_modify_regiments = yes
	exists = var:maa_manager_selected_regiment_scope
	var:maa_manager_selected_regiment_scope = { maa_manager_is_manageable_regiment = yes }
	exists = scope:target_regiment
	scope:target_regiment = { maa_manager_compatible_target = yes }
}

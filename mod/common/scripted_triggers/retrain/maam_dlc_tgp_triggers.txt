# MAA Manager - DLC TGP (The Grand Path) MAA Recruitment Triggers
#
# Contains 17 new MAA types from The Grand Path DLC (10_tgp_maa_types.txt)
# East Asian cultural MAA + siege weapons for Chinese/Korean/Japanese heritages
#
# Cultural MAA (12): emishi_horse_archers, japanese_horse_archers, warrior_monks,
#                     fire_lancers, gakgung_archers, pesilat_warriors, samurai,
#                     zhanmadao_infantry, tiefutu, burenjia, rocket_carts,
#                     black_armor_cavalry (v1.19)
# Regional MAA (1): ballista_elephant
# Siege Weapons (4): ballista, cloud_ladder, siege_tower, cannon

# ===== CULTURAL MAA =====

maa_manager_can_recruit_emishi_horse_archers = {
	trigger_if = {
		limit = { maa_manager_is_title_mode = yes }
		maa_manager_can_recruit_title_type = { TYPE = emishi_horse_archers }
	}
	trigger_else = {
		# Requires Emishi cultural tradition (Japanese)
		culture = {
			has_cultural_parameter = unlock_emishi_horse_archers_units
		}
	}
}

maa_manager_can_recruit_fire_lancers = {
	trigger_if = {
		limit = { maa_manager_is_title_mode = yes }
		maa_manager_can_recruit_title_type = { TYPE = fire_lancers }
	}
	trigger_else = {
		# Requires Fire Lancers cultural tradition + gunpowder building limit
		# Similar to handgunners - needs powder magazine support
		# (a gunpowder source being replaced frees its own quota slot)
		trigger_if = {
			limit = { maa_manager_retrain_replaces_gunpowder_unit = yes }
			gunpowder_raise_limit_existing_gunpowder_units <= gunpowder_raise_limit_existing_gunpowder_buildings
		}
		trigger_else = {
			gunpowder_raise_limit_existing_gunpowder_units < gunpowder_raise_limit_existing_gunpowder_buildings
		}
		culture = {
			has_cultural_parameter = unlock_fire_lancers_units
		}
	}
}

		maa_manager_can_recruit_title_type = { TYPE = zhanmadao_infantry }
	}
	trigger_else = {
		# Requires Chinese cultural tradition + military/protectorate province contract or independence
		valid_for_maa_trigger = { PARAMETER = unlock_zhanmadao }
		OR = {
			vassal_contract_has_flag = celestial_province_military
			vassal_contract_has_flag = celestial_province_protectorate
			is_independent_ruler = yes
		}
		# Excluded by Strength in Numbers tradition (bans heavy MAA)
		NOT = {
			culture = {
				has_cultural_parameter = strength_in_numbers_heavy_maa_ban
			}
		}
	}
}

# ===== REGIONAL MAA =====

maa_manager_can_recruit_ballista_elephant = {
	trigger_if = {
		limit = { maa_manager_is_title_mode = yes }
		maa_manager_can_recruit_title_type = { TYPE = ballista_elephant }
	}
	trigger_else = {
		# Requires elephantry + advanced bowmaking innovations
		# Excluded for landless adventurers
		culture ?= {
			has_innovation = innovation_elephantry
			has_innovation = innovation_advanced_bowmaking
		}
		NOT = {
			is_landless_adventurer = yes
		}
		# Excluded by Strength in Numbers tradition (bans heavy MAA, v1.20)
		NOT = {
			culture = {
				has_cultural_parameter = strength_in_numbers_heavy_maa_ban
			}
		}
	}
}

# ===== SIEGE WEAPONS (East Asian) =====

		limit = { maa_manager_is_title_mode = yes }
		maa_manager_can_recruit_title_type = { TYPE = ballista }
	}
	trigger_else = {
		# East Asian replacement for catapult
		# Requires catapult innovation + East Asian heritage
		culture = {
			has_innovation = innovation_catapult
			culture_uses_eastern_siege_weapons_trigger = yes
		}
		# Excluded for steppe governments (nomadic and herder)
		NOT = { government_has_flag = government_is_in_steppe }

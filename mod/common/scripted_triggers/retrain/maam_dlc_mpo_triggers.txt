######################################################
# MAA Manager - DLC MPO (Roads to Power) Nomadic MAA Recruitment Triggers
#
# Contains 8 Nomadic DLC MAA types (09_mpo_maa_types.txt)
# Nomadic government and cultural tradition-gated types
#
# Types: torch_bearers, kheshig, steppe_raiders, heavy_horse_archers,
#        nomad_lancers, mangudai, cataphract_archers, maturkan_warriors
######################################################

maa_manager_can_recruit_torch_bearers = {
	trigger_if = {
		limit = { maa_manager_is_title_mode = yes }
		maa_manager_can_recruit_title_type = { TYPE = torch_bearers }
	}
	trigger_else = {
		# Siege weapon for nomadic government
		government_has_flag = government_is_nomadic
	}
}

maa_manager_can_recruit_kheshig = {
	trigger_if = {
		limit = { maa_manager_is_title_mode = yes }
		maa_manager_can_recruit_title_type = { TYPE = kheshig }
	}
	trigger_else = {
		# Only one regiment of this type allowed (max_regiments = 1)
		NOT = {
			any_maa_regiment = {
				is_maa_type = kheshig
			}
		}
		# Requires dynasty with mpo_nomad_legacy_4 perk
		dynasty ?= {
			has_dynasty_perk = mpo_nomad_legacy_4
		}
	}
}

maa_manager_can_recruit_steppe_raiders = {
	trigger_if = {
		limit = { maa_manager_is_title_mode = yes }
		maa_manager_can_recruit_title_type = { TYPE = steppe_raiders }
	}
	trigger_else = {
		# Use game's native trigger directly
		mpo_can_recruit_nomad_maa_trigger = yes
	}
}

maa_manager_can_recruit_heavy_horse_archers = {
	trigger_if = {
		limit = { maa_manager_is_title_mode = yes }
		maa_manager_can_recruit_title_type = { TYPE = heavy_horse_archers }
	}
	trigger_else = {
		# Use game's native trigger plus domicile check
		mpo_can_recruit_nomad_maa_trigger = yes
		trigger_if = {
			limit = {
				government_has_flag = government_is_nomadic
				exists = domicile
			}
			domicile = {
				has_domicile_parameter = unlock_maa_heavy_horse_archers
			}
		}
	}
}

maa_manager_can_recruit_nomad_lancers = {
	trigger_if = {
		limit = { maa_manager_is_title_mode = yes }
		maa_manager_can_recruit_title_type = { TYPE = nomad_lancers }
	}
	trigger_else = {
		# Use game's native trigger directly
		mpo_can_recruit_nomad_heavy_cavalry_maa_trigger = yes
		# Excluded by Strength in Numbers tradition (bans heavy MAA)
		NOT = {
			culture = {
				has_cultural_parameter = strength_in_numbers_heavy_maa_ban
			}
		}
	}
}

maa_manager_can_recruit_mangudai = {
	trigger_if = {
		limit = { maa_manager_is_title_mode = yes }
		maa_manager_can_recruit_title_type = { TYPE = mangudai }
	}
	trigger_else = {
		# Requires Mongolic cultural tradition: unlock_maa_mangudai
		valid_for_maa_trigger = { PARAMETER = unlock_maa_mangudai }
	}
}

maa_manager_can_recruit_cataphract_archers = {
	trigger_if = {
		limit = { maa_manager_is_title_mode = yes }
		maa_manager_can_recruit_title_type = { TYPE = cataphract_archers }
	}
	trigger_else = {
		# Requires cultural tradition + either non-nomadic OR has dominance
		valid_for_maa_trigger = { PARAMETER = unlock_maa_cataphract_archers }
		# Excluded by Strength in Numbers tradition (bans heavy MAA)
		NOT = {
			culture = {
				has_cultural_parameter = strength_in_numbers_heavy_maa_ban
			}
		}
		OR = {
			NOT = {
				government_has_flag = government_is_nomadic
			}
			has_realm_law_flag = unlock_maa_nomad_lancers
		}
	}
}

maa_manager_can_recruit_maturkan_warriors = {
	trigger_if = {
		limit = { maa_manager_is_title_mode = yes }
		maa_manager_can_recruit_title_type = { TYPE = maturkan_warriors }
	}
	trigger_else = {
		# Requires Caucasian cultural tradition: unlock_maa_maturkan_warriors
		valid_for_maa_trigger = { PARAMETER = unlock_maa_maturkan_warriors }
	}
}

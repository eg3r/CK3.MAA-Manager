######################################################
# MAA Manager - DLC FP1 (Northern Lords) MAA Recruitment Triggers
#
# Contains 4 Northern Lords DLC MAA types (00_fp1_maa_types.txt)
# Cultural tradition-gated types available to Norse cultures
#
# Types: vigmen, bondi, varangian_veterans, jomsviking_pirates (blocked)
######################################################

maa_manager_can_recruit_vigmen = {
	trigger_if = {
		limit = { maa_manager_is_title_mode = yes }
		maa_manager_can_recruit_title_type = { TYPE = vigmen }
	}
	trigger_else = {
		valid_for_maa_trigger = { PARAMETER = unlock_maa_vigmen }
	}
}

maa_manager_can_recruit_bondi = {
	trigger_if = {
		limit = { maa_manager_is_title_mode = yes }
		maa_manager_can_recruit_title_type = { TYPE = bondi }
	}
	trigger_else = {
		valid_for_maa_trigger = { PARAMETER = unlock_maa_bondi }
	}
}

maa_manager_can_recruit_varangian_veterans = {
	trigger_if = {
		limit = { maa_manager_is_title_mode = yes }
		maa_manager_can_recruit_title_type = { TYPE = varangian_veterans }
	}
	trigger_else = {
		# Only one regiment of this type allowed (max_regiments = 1)
		NOT = {
			any_maa_regiment = {
				is_maa_type = varangian_veterans
			}
		}
		valid_for_maa_trigger = { PARAMETER = unlock_maa_varangian_veterans }
		NOT = {
			culture = {
				has_cultural_parameter = strength_in_numbers_heavy_maa_ban
			}
		}
	}
}

maa_manager_can_recruit_jomsviking_pirates = {
	trigger_if = {
		limit = { maa_manager_is_title_mode = yes }
		maa_manager_can_recruit_title_type = { TYPE = jomsviking_pirates }
	}
	trigger_else = {
		always = no	# Not manually recruitable in base game
	}
}

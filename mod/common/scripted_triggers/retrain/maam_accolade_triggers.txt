######################################################
# MAA Manager - Accolade MAA Recruitment Triggers
#
# Contains 11 accolade MAA types (01_accolade_maa_types.txt)
# Mirrors the game's can_recruit (since v1.19): any accolade whose
# attribute of the matching type is at level 2 or higher.
# Only 1 regiment of each type allowed per character (max_regiments = 1)
#
# The game's "NOT = { exists = scope:title }" clause is omitted:
# the manager only creates personal (non-title) regiments.
#
# Types: archers, skirmishers, pikes, outriders, vanguards, lancers,
#        crossbowers, cameliers, elephantiers, horse_archers_retinue,
#        accolade_maa_handgunners
######################################################

maa_manager_can_recruit_archers = {
	# Only one regiment of this type allowed
	NOT = {
		any_maa_regiment = {
			is_maa_type = accolade_maa_archers
		}
	}
	# Requires an accolade with the attribute at level 2+
	any_accolade = {
		accolade_attribute_level = {
			type = archer_attribute
			level >= 2
		}
	}
}

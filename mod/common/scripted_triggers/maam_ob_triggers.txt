######################################################
# MAA Manager - Order of battle (Military window board) triggers
# ROOT = the player. Regiment scopes are passed by the GUI.
######################################################

# Regiments may only be changed at peace with nothing raised (checked again by
# every action, not only by the GUI).
maam_can_modify_regiments = {
	is_at_war = no
	has_raised_armies = no
	NOT = {
		any_maa_regiment = {
			is_raised = yes
		}
	}
}

# Regiment scope: a title regiment the player owns through a held title.
maam_title_regiment_owned = {
	is_title_maa_regiment = yes
	exists = regiment_owning_title
	regiment_owning_title = { holder = root }
}

# Regiment scope: an owned title regiment that is home, unraised and controlled.
maam_title_regiment_manageable = {
	maam_title_regiment_owned = yes
	is_raised = no
	exists = regiment_controlling_title
	regiment_controlling_title = regiment_owning_title
	regiment_controller = root
}

# Board orders may run: peace, nothing raised, no retrain open.
maam_ob_can_order = {
	maam_can_modify_regiments = yes
	NOT = { has_variable = maam_rt_active }
}

# The open order is of kind KIND (lift, ended, merge, pour).
maam_ob_kind_is = {
	exists = var:maam_ob_kind
	var:maam_ob_kind = flag:$KIND$
}

# Regiment scope: may take part in board orders (independent of title mode).
maam_ob_regiment_ok = {
	OR = {
		AND = {
			is_personal_maa_regiment = yes
			regiment_owner = root
			is_raised = no
		}
		maam_title_regiment_manageable = yes
	}
}

# Regiment scope: on the same board as $OTHER$ (both personal, or the same title).
maam_ob_same_board = {
	trigger_if = {
		limit = { is_title_maa_regiment = yes }
		$OTHER$ = { is_title_maa_regiment = yes }
		regiment_owning_title = $OTHER$.regiment_owning_title
	}
	trigger_else = {
		$OTHER$ = { is_title_maa_regiment = no }
	}
}

# The stored order source still exists and may be managed.
maam_ob_src_ok = {
	exists = var:maam_ob_src
	var:maam_ob_src = { maam_ob_regiment_ok = yes }
}

# scope:target_regiment can take part with the order source. Type equality is
# the vanilla list or the GUI's ObjectsEqual passed as scope:type_relation
# (the GUI is the authority in single player), so modded types work too.
maam_ob_kin_target = {
	exists = scope:target_regiment
	maam_ob_src_ok = yes
	scope:target_regiment = {
		maam_ob_regiment_ok = yes
		NOT = { this = root.var:maam_ob_src }
		maam_ob_same_board = { OTHER = root.var:maam_ob_src }
		OR = {
			maam_same_type_as = { OTHER = root.var:maam_ob_src }
			AND = {
				exists = scope:type_relation
				scope:type_relation = flag:same
			}
		}
	}
}

# A receipt must still describe both regiments exactly. Reinforcement or a
# native resize invalidates exact undo; zero transferred soldiers is valid.
maam_ob_history_current = {
	has_variable = maam_hist_version
	var:maam_hist_version = 2
	exists = var:maam_hist_1
	exists = var:maam_hist_1_src
	var:maam_hist_1_src = var:maam_ob_src
	has_variable = maam_hist_1_men
	has_variable = maam_hist_1_src_size
	has_variable = maam_hist_1_src_troops
	has_variable = maam_hist_1_src_max
	has_variable = maam_hist_1_dst_size
	has_variable = maam_hist_1_dst_troops
	has_variable = maam_hist_1_dst_max
	has_variable = maam_hist_1_src_max_before
	has_variable = maam_hist_1_dst_max_before
	maam_ob_src_ok = yes
	var:maam_ob_src = {
		regiment_controller = root
		maa_size = root.var:maam_hist_1_src_size
		maa_current_troops_count = root.var:maam_hist_1_src_troops
		maa_max_troops_count = root.var:maam_hist_1_src_max
	}
	var:maam_hist_1 = {
		maam_ob_regiment_ok = yes
		regiment_controller = root
		maa_size > 1
		NOT = { this = root.var:maam_ob_src }
		maam_ob_same_board = { OTHER = root.var:maam_ob_src }
		maa_size = root.var:maam_hist_1_dst_size
		maa_current_troops_count = root.var:maam_hist_1_dst_troops
		maa_max_troops_count = root.var:maam_hist_1_dst_max
	}
}

# "Move a stack back" also needs the native displayed size limit of its
# receiver (the original source), and GUI type equality for additive types.
maam_ob_back_ok = {
	maam_ob_history_current = yes
	maam_ob_gui_room = yes
	exists = scope:type_relation
	scope:type_relation = flag:same
	var:maam_ob_src = { can_upgrade_maa = yes }
	var:maam_hist_1_men >= 0
	var:maam_hist_1_dst_troops >= var:maam_hist_1_men
	var:maam_hist_1_src_max_before >= { value = var:maam_hist_1_src_troops add = var:maam_hist_1_men }
	var:maam_hist_1_dst_max_before >= { value = var:maam_hist_1_dst_troops subtract = var:maam_hist_1_men }
}

# The soldier move has a valid partner (maam_ob_tgt) for its regiment (maam_ob_src).
maam_ob_pour_pair_ok = {
	maam_ob_kind_is = { KIND = pour }
	maam_ob_src_ok = yes
	exists = var:maam_ob_tgt
	var:maam_ob_tgt = {
		maam_ob_regiment_ok = yes
		NOT = { this = root.var:maam_ob_src }
		maam_ob_same_board = { OTHER = root.var:maam_ob_src }
	}
}

# Soldiers can move from $FROM$ to $TO$: the sender has more than its first
# stack, the receiver has room (each move: at most one stack).
maam_ob_can_send = {
	$FROM$ = { maa_current_troops_count > maam_ob_stack_value }
	$TO$ = { maa_current_troops_count < maa_max_troops_count }
}

# Quartermaster (character scope): the stored tip still holds. Grow: the
# regiment still has room (it may have been grown in its own window). Combine:
# both regiments are still there and every slot is still used.
maam_qm_tip_valid = {
	exists = var:maam_qm_kind
	exists = var:maam_qm_a
	var:maam_qm_a = { maam_ob_regiment_ok = yes }
	trigger_if = {
		limit = { var:maam_qm_kind = flag:grow }
		var:maam_qm_a = { can_upgrade_maa = yes }
	}
	trigger_else = {
		exists = var:maam_qm_b
		var:maam_qm_b = { maam_ob_regiment_ok = yes }
		maa_regiments_count >= maa_regiments_max_count
	}
}

# A stack move's click passes the size limit the game shows (Regiment.
# CalcMaxMAAChunkCount, "Size X/Y") as scope:gui_room. can_upgrade_maa alone
# let title regiments grow past the title limit.
maam_ob_gui_room = {
	exists = scope:gui_room
	scope:gui_room = flag:yes
}

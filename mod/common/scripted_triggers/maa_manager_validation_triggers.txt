######################################################
# MAA Manager - Validation Triggers
######################################################

######################################################
# Regiments the manager may list, move and retrain
# Personal mode accepts owned personal troops; title mode additionally
# requires the owning title to retain control. Loans remain read-only.
# Scope: regiment
######################################################
maa_manager_is_manageable_regiment = {
	trigger_if = {
		limit = { root = { maa_manager_is_title_mode = yes } }
		maa_manager_title_regiment_manageable = yes
	}
	trigger_else = {
		is_personal_maa_regiment = yes
		regiment_owner = root
	}
}

######################################################
# Manager policy: regiments may only be changed at peace with nothing raised
# (same rule as the button that opens the manager window).
# Checked again by every mutating action, not only when the window opens.
# Scope: character
######################################################
maa_manager_can_modify_regiments = {
	is_at_war = no
	has_raised_armies = no
	NOT = {
		any_maa_regiment = {
			is_raised = yes
		}
	}
}

######################################################
# Live check of the selected transfer source and target
# Both must still be owned, manageable, distinct and of the same type.
# Title mode also rechecks their shared owning title and current control.
# Scope: character (root)
######################################################
maa_manager_transfer_selection_valid = {
	exists = var:maa_manager_selected_regiment_scope
	exists = var:maa_manager_selected_target_scope
	var:maa_manager_selected_regiment_scope = { maa_manager_is_manageable_regiment = yes }
	var:maa_manager_selected_target_scope = { maa_manager_compatible_target = yes }
}

######################################################
# Retrain target is paid in piety
# Warrior Monks are the only MAA the game recruits with piety
# (buy_cost = { piety = warrior_monks_piety_recruitment_cost }).
# Scope: character
######################################################
maa_manager_retrain_pays_piety = {
	has_variable = maa_manager_retrain_target_type
	var:maa_manager_retrain_target_type = flag:warrior_monks
}

######################################################
# The retrain source is a gunpowder regiment counted by the native quota
# (gunpowder_raise_limit_existing_gunpowder_units counts personal handgunners
# and fire lancers). Replacing it frees its quota slot, so the quota check for
# a gunpowder target may allow one more.
# Scope: character
######################################################
maa_manager_retrain_replaces_gunpowder_unit = {
	has_variable = maa_manager_selected_regiment_scope
	var:maa_manager_selected_regiment_scope = {
		is_personal_maa_regiment = yes
		OR = {
			is_maa_type = handgunners
			is_maa_type = fire_lancers
		}
	}
}

######################################################
# Retrain affordability
# Warrior Monks target: piety (their native currency).
# Tribal: prestige. Everyone else: treasury_or_gold, which checks the
# treasury for treasury governments (replace_gold_cost_by_treasury) and
# gold otherwise - the same resource the game charges for MAA recruitment.
# Scope: character
######################################################
maa_manager_can_afford_retrain = {
	trigger_if = {
		limit = { maa_manager_retrain_pays_piety = yes }
		piety >= maa_manager_retrain_piety_cost
	}
	trigger_else_if = {
		limit = { has_government = tribal_government }
		prestige >= maa_manager_retrain_prestige_cost
	}
	trigger_else_if = {
		limit = { maa_manager_is_title_mode = yes }
		trigger_if = {
			limit = { maa_manager_title_uses_treasury = yes }
			treasury >= maa_manager_retrain_gold_cost
		}
		trigger_else = { gold >= maa_manager_retrain_gold_cost }
	}
	trigger_else = { treasury_or_gold >= maa_manager_retrain_gold_cost }
}

######################################################
# Full confirm-time check for maa_manager_execute_retrain
# Used by is_valid AND as the gate inside the effect, so a stale selection
# can never charge the player or destroy the source regiment.
# Scope: character (root)
######################################################
maa_manager_can_execute_retrain = {
	maa_manager_can_modify_regiments = yes
	has_variable = maa_manager_selected_regiment_scope
	has_variable = maa_manager_retrain_target_type
	exists = var:maa_manager_selected_regiment_scope
	var:maa_manager_selected_regiment_scope = { maa_manager_is_manageable_regiment = yes }
	trigger_if = {
		limit = { maa_manager_is_title_mode = yes }
		maa_manager_title_retrain_source_valid = yes
	}
	# Target must still pass its recruit trigger (culture, government, accolades, ...)
	maa_manager_retrain_target_is_recruitable = yes
	maa_manager_can_afford_retrain = yes
}

# MAA Manager - Helper Effects for Scripted GUIs
######################################################

######################################################
# Full data refresh and state reset (the single reset policy)
# Clears the source and target selection, planned transfers, cached
# capacities and retrain state, then rebuilds the regiment list.
# Scope: character
######################################################
maa_manager_data_create_effect = {
	# Clear any previously collected data to ensure a fresh list
	clear_variable_list = maa_regiment_list
	clear_variable_list = maa_move_target_list
	clear_saved_scope = maa_manager_selected_regiment
	save_scope_value_as = {
		name = maa_manager_selected_type
		value = flag:_maa_manager_unset
	}
	remove_variable = maa_manager_move_has_targets

	# Clear transfer state (target, planned amounts, cached capacities)
	remove_variable = maa_manager_selected_target_scope
	remove_variable = maa_total_planned_transfers
	remove_variable = maa_total_planned_regiment_transfers
	remove_variable = maa_target_capacity
	remove_variable = maa_target_regiment_capacity

	# Clear retrain-specific variables to ensure clean state
	remove_variable = maa_manager_selected_regiment_scope
	remove_variable = maa_manager_retrain_target_type
	remove_variable = maa_manager_target_cost
	remove_variable = maa_manager_source_cost

	# Iterate through all of the character's personal MAA regiments
	# (title MAA and other non-personal regiments excluded - see maa_manager_is_manageable_regiment)
	every_maa_regiment = {
		limit = { maa_manager_is_manageable_regiment = yes }
		ROOT = {
			add_to_variable_list = {
				name = maa_regiment_list
				target = prev
			}
		}
	}
}

######################################################
# Full refresh that keeps the current source selected
# Used after a transfer when the source regiment still exists, so the
# player can continue working with it. Everything else is reset.
# Scope: character
######################################################
maa_manager_refresh_keep_source_effect = {
	if = {
		limit = { has_variable = maa_manager_selected_regiment_scope }
		var:maa_manager_selected_regiment_scope = {
			save_scope_as = maa_manager_reselect_regiment
		}
	}
	maa_manager_data_create_effect = yes
	if = {
		limit = {
			exists = scope:maa_manager_reselect_regiment
			any_maa_regiment = {
				this = scope:maa_manager_reselect_regiment
				maa_manager_is_manageable_regiment = yes
			}
		}
		scope:maa_manager_reselect_regiment = {
			save_scope_as = regiment
		}
		maa_manager_select_regiment_effect = yes
	}
}

maa_manager_select_regiment_effect = {
	clear_variable_list = maa_move_target_list
	remove_variable = maa_manager_move_has_targets
	scope:regiment = {
		save_scope_as = maa_manager_selected_regiment
		ROOT = {
			set_variable = {
				name = maa_manager_selected_regiment_scope
				value = scope:regiment
			}
		}
		maa_manager_store_selected_type = yes
	}
	every_maa_regiment = {
		if = {
			limit = {
				NOT = {
					this = scope:maa_manager_selected_regiment
				}
				maa_manager_is_manageable_regiment = yes
				maa_manager_type_matches_selection = yes
			}
			ROOT = {
				add_to_variable_list = {
					name = maa_move_target_list
					target = prev
				}
				set_variable = {
					name = maa_manager_move_has_targets
					value = yes
				}
			}
		}
	}
}

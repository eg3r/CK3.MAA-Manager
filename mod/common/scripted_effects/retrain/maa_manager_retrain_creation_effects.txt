# Shared literal type dispatch. Personal creation retains its existing behavior.
maa_manager_create_retrain_regiment_effect = {
	if = {
		limit = { maa_manager_is_title_mode = yes }
		create_maa_regiment = {
			type = $TYPE$
			title = scope:maa_manager_owning_title
			check_can_recruit = yes
			size = 1
		}
	}
	else = {
		create_maa_regiment = { type = $TYPE$ check_can_recruit = no size = 1 }
	}
}

# Character scope; DONOR/RECEIVER are regiment scope expressions.
# Expand and verify the receiver before taking anything from the donor.
maa_manager_transfer_title_stack_effect = {
	if = {
		limit = {
			maa_manager_is_title_mode = yes
			maa_manager_can_modify_regiments = yes
			maa_manager_transfer_selection_valid = yes
			$RECEIVER$ = { can_upgrade_maa = yes }
		}
		set_local_variable = { name = maa_title_donor_size value = $DONOR$.maa_size }
		set_local_variable = { name = maa_title_receiver_size value = $RECEIVER$.maa_size }
		set_local_variable = { name = maa_title_donor_alive value = $DONOR$.maa_current_troops_count }
		set_local_variable = {
			name = maa_title_donor_chunk
			value = { value = $DONOR$.maa_max_troops_count divide = $DONOR$.maa_size }
		}
		set_local_variable = {
			name = maa_title_donor_remaining_capacity
			value = { value = $DONOR$.maa_max_troops_count subtract = local_var:maa_title_donor_chunk }
		}
		$RECEIVER$ = { change_maa_regiment_size = { size = 1 reinforce = no } }
		set_local_variable = {
			name = maa_title_soldiers
			value = {
				value = local_var:maa_title_donor_alive
				max = local_var:maa_title_donor_chunk
				max = { value = $RECEIVER$.maa_max_troops_count subtract = $RECEIVER$.maa_current_troops_count }
				min = 0
			}
		}
		if = {
			limit = {
				$RECEIVER$ = { maa_size = { value = local_var:maa_title_receiver_size add = 1 } }
				# Shrinking must not discard soldiers that cannot fit in the receiver.
				local_var:maa_title_donor_remaining_capacity >= {
					value = local_var:maa_title_donor_alive
					subtract = local_var:maa_title_soldiers
				}
			}
			$DONOR$ = {
				change_maa_troops_count = { value = local_var:maa_title_soldiers multiply = -1 }
			}
			$RECEIVER$ = { change_maa_troops_count = local_var:maa_title_soldiers }
			if = {
				limit = { local_var:maa_title_donor_size = 1 }
				$DONOR$ = { destroy_maa_regiment = yes }
			}
			else = { $DONOR$ = { change_maa_regiment_size = { size = -1 reinforce = no } } }
		}
		else = {
			# No soldiers have moved. Undo only the receiver's added capacity.
			$RECEIVER$ = {
				change_maa_regiment_size = {
					size = { value = local_var:maa_title_receiver_size subtract = maa_size }
					reinforce = no
				}
			}
		}
		if = {
			limit = {
				exists = var:maa_manager_selected_regiment_scope
				exists = var:maa_manager_selected_target_scope
			}
			# Keep both selections so the existing +/- controls can be clicked again.
			set_variable = { name = maa_total_planned_transfers value = 0 }
			set_variable = { name = maa_total_planned_regiment_transfers value = 0 }
			set_variable = {
				name = maa_target_capacity
				value = {
					value = var:maa_manager_selected_target_scope.maa_max_troops_count
					subtract = var:maa_manager_selected_target_scope.maa_current_troops_count
				}
			}
			set_variable = {
				name = maa_target_regiment_capacity
				value = var:maa_manager_selected_target_scope.maa_size
			}
		}
		else = { maa_manager_refresh_keep_source_effect = yes }
	}
}

######################################################
# MAA Manager - Muster board effects (selection, stacks, merge, soldiers)
#
# Stack moves are immediate facts. Give/Take NEVER move a donor's last stack:
# a fast double click sends two commands and the second must be refused here,
# not only hidden in the GUI. Merging the last stack is its own action.
# Scope: character (ROOT = player). Regiment selection lives in the existing
# maa_manager_selected_regiment_scope / maa_manager_selected_target_scope.
######################################################

# Select scope:regiment for the board. Title regiments switch to title mode,
# personal regiments to personal mode, so all existing title rules apply.
maam_board_select_effect = {
	remove_variable = maam_preselect_regiment
	if = {
		limit = { exists = scope:regiment scope:regiment = { is_title_maa_regiment = yes } }
		set_variable = { name = maa_manager_title_mode value = yes }
	}
	else = { remove_variable = maa_manager_title_mode }
	remove_variable = maam_gui_same_target
	remove_variable = maam_gui_same_source
	remove_variable = maam_last_from
	remove_variable = maam_last_to
	remove_variable = maam_board_loaned
	maa_manager_data_create_effect = yes
	if = {
		limit = { maa_manager_select_source_valid = yes }
		maa_manager_select_regiment_effect = yes
		set_variable = { name = maam_board_selected value = scope:regiment }
		# Exactly one known partner: pair it straight away.
		if = {
			limit = { variable_list_size = { name = maa_move_target_list target = 1 } }
			every_in_list = {
				variable = maa_move_target_list
				save_scope_as = target_regiment
			}
			maam_board_set_partner_effect = yes
		}
	}
	else = {
		remove_variable = maam_board_selected
		# Owned but loaned out: remember it for the read-only notice.
		if = {
			limit = {
				exists = scope:regiment
				scope:regiment = { maa_manager_title_regiment_owned = yes }
			}
			set_variable = { name = maam_board_loaned value = scope:regiment }
		}
	}
}

# Pair scope:target_regiment (optional scope:type_relation = flag:same).
maam_board_set_partner_effect = {
	if = {
		limit = { exists = scope:type_relation scope:type_relation = flag:same }
		set_variable = { name = maam_gui_same_target value = scope:target_regiment }
		set_variable = { name = maam_gui_same_source value = var:maa_manager_selected_regiment_scope }
	}
	if = {
		limit = { maa_manager_select_target_valid = yes }
		set_variable = { name = maa_manager_selected_target_scope value = scope:target_regiment }
		remove_variable = maam_last_from
		remove_variable = maam_last_to
		maam_board_reset_plan_effect = yes
	}
}

maam_board_clear_partner_effect = {
	remove_variable = maa_manager_selected_target_scope
	remove_variable = maam_gui_same_target
	remove_variable = maam_gui_same_source
	remove_variable = maam_last_from
	remove_variable = maam_last_to
	maam_board_reset_plan_effect = yes
}

maam_board_clear_effect = {
	maam_board_clear_partner_effect = yes
	remove_variable = maam_board_selected
	remove_variable = maa_manager_selected_regiment_scope
	clear_variable_list = maa_move_target_list
	remove_variable = maa_manager_move_has_targets
}

# Zeroes the soldier plan and refreshes the cached capacities.
maam_board_reset_plan_effect = {
	set_variable = { name = maa_total_planned_transfers value = 0 }
	set_variable = { name = maa_total_planned_regiment_transfers value = 0 }
	if = {
		limit = { exists = var:maa_manager_selected_target_scope }
		set_variable = {
			name = maa_target_capacity
			value = {
				value = var:maa_manager_selected_target_scope.maa_max_troops_count
				subtract = var:maa_manager_selected_target_scope.maa_current_troops_count
			}
		}
	}
	if = {
		limit = { exists = var:maa_manager_selected_regiment_scope }
		set_variable = {
			name = maa_stack_size
			value = {
				value = var:maa_manager_selected_regiment_scope.maa_max_troops_count
				divide = var:maa_manager_selected_regiment_scope.maa_size
			}
		}
	}
}

# Core stack move for both personal and title regiments. Expands and verifies
# the receiver before anything leaves the donor; reverts the receiver if the
# donor's remaining soldiers would not fit. Destroys the donor only when it had
# exactly one stack (callers decide whether that is allowed).
maam_stack_core_effect = {
	set_local_variable = { name = maam_donor_size value = $DONOR$.maa_size }
	set_local_variable = { name = maam_receiver_size value = $RECEIVER$.maa_size }
	set_local_variable = { name = maam_donor_alive value = $DONOR$.maa_current_troops_count }
	set_local_variable = {
		name = maam_donor_chunk
		value = { value = $DONOR$.maa_max_troops_count divide = $DONOR$.maa_size }
	}
	set_local_variable = {
		name = maam_donor_remaining_capacity
		value = { value = $DONOR$.maa_max_troops_count subtract = local_var:maam_donor_chunk }
	}
	$RECEIVER$ = { change_maa_regiment_size = { size = 1 reinforce = no } }
	set_local_variable = {
		name = maam_soldiers
		value = {
			value = local_var:maam_donor_alive
			max = local_var:maam_donor_chunk
			max = { value = $RECEIVER$.maa_max_troops_count subtract = $RECEIVER$.maa_current_troops_count }
			min = 0
		}
	}
	# The last stack takes every soldier with it.
	if = {
		limit = { local_var:maam_donor_size = 1 }
		set_local_variable = {
			name = maam_soldiers
			value = {
				value = local_var:maam_donor_alive
				max = { value = $RECEIVER$.maa_max_troops_count subtract = $RECEIVER$.maa_current_troops_count }
				min = 0
			}
		}
	}
	if = {
		limit = {
			$RECEIVER$ = { maa_size = { value = local_var:maam_receiver_size add = 1 } }
			trigger_if = {
				limit = { local_var:maam_donor_size > 1 }
				local_var:maam_donor_remaining_capacity >= {
					value = local_var:maam_donor_alive
					subtract = local_var:maam_soldiers
				}
			}
			trigger_else = { local_var:maam_soldiers >= local_var:maam_donor_alive }
		}
		$DONOR$ = { change_maa_troops_count = { value = local_var:maam_soldiers multiply = -1 } }
		$RECEIVER$ = { change_maa_troops_count = local_var:maam_soldiers }
		if = {
			limit = { local_var:maam_donor_size = 1 }
			$DONOR$ = { destroy_maa_regiment = yes }
		}
		else = { $DONOR$ = { change_maa_regiment_size = { size = -1 reinforce = no } } }
		set_local_variable = { name = maam_moved value = yes }
	}
	else = {
		# Nothing left the donor; undo only the receiver's added capacity.
		$RECEIVER$ = {
			change_maa_regiment_size = {
				size = { value = local_var:maam_receiver_size subtract = maa_size }
				reinforce = no
			}
		}
	}
}

# Selected -> partner, never the last stack.
maam_board_give_effect = {
	if = {
		limit = {
			maam_board_actions_allowed = yes
			maa_manager_transfer_selection_valid = yes
			var:maa_manager_selected_regiment_scope = { maa_size > 1 }
			var:maa_manager_selected_target_scope = { can_upgrade_maa = yes }
		}
		maam_stack_core_effect = {
			DONOR = root.var:maa_manager_selected_regiment_scope
			RECEIVER = root.var:maa_manager_selected_target_scope
		}
		if = {
			limit = { has_local_variable = maam_moved }
			set_variable = { name = maam_last_from value = var:maa_manager_selected_regiment_scope }
			set_variable = { name = maam_last_to value = var:maa_manager_selected_target_scope }
		}
		maam_board_reset_plan_effect = yes
	}
}

# Partner -> selected, never the partner's last stack.
maam_board_take_effect = {
	if = {
		limit = {
			maam_board_actions_allowed = yes
			maa_manager_transfer_selection_valid = yes
			var:maa_manager_selected_target_scope = { maa_size > 1 }
			var:maa_manager_selected_regiment_scope = { can_upgrade_maa = yes }
		}
		maam_stack_core_effect = {
			DONOR = root.var:maa_manager_selected_target_scope
			RECEIVER = root.var:maa_manager_selected_regiment_scope
		}
		if = {
			limit = { has_local_variable = maam_moved }
			set_variable = { name = maam_last_from value = var:maa_manager_selected_target_scope }
			set_variable = { name = maam_last_to value = var:maa_manager_selected_regiment_scope }
		}
		maam_board_reset_plan_effect = yes
	}
}

# Reverse the last stack move (another immediate move, never a last stack).
maam_board_undo_effect = {
	if = {
		limit = {
			maam_board_actions_allowed = yes
			maa_manager_transfer_selection_valid = yes
			maam_board_undo_pair_current = yes
			var:maam_last_to = { maa_size > 1 }
			var:maam_last_from = { can_upgrade_maa = yes }
		}
		maam_stack_core_effect = {
			DONOR = root.var:maam_last_to
			RECEIVER = root.var:maam_last_from
		}
		remove_variable = maam_last_from
		remove_variable = maam_last_to
		maam_board_reset_plan_effect = yes
	}
}

# Merge the selected regiment's LAST stack into the partner (selected disbands).
maam_board_merge_into_partner_effect = {
	if = {
		limit = {
			maam_board_actions_allowed = yes
			maa_manager_transfer_selection_valid = yes
			var:maa_manager_selected_regiment_scope = { maa_size = 1 }
			var:maa_manager_selected_target_scope = { can_upgrade_maa = yes }
		}
		set_local_variable = { name = maam_survivor value = var:maa_manager_selected_target_scope }
		maam_stack_core_effect = {
			DONOR = root.var:maa_manager_selected_regiment_scope
			RECEIVER = root.var:maa_manager_selected_target_scope
		}
		if = {
			limit = { NOT = { exists = var:maa_manager_selected_regiment_scope } }
			# The survivor becomes the selected regiment.
			local_var:maam_survivor = { save_scope_as = regiment }
			maam_board_select_effect = yes
		}
	}
}

# Merge the partner's LAST stack into the selected regiment (partner disbands).
maam_board_merge_partner_in_effect = {
	if = {
		limit = {
			maam_board_actions_allowed = yes
			maa_manager_transfer_selection_valid = yes
			var:maa_manager_selected_target_scope = { maa_size = 1 }
			var:maa_manager_selected_regiment_scope = { can_upgrade_maa = yes }
		}
		maam_stack_core_effect = {
			DONOR = root.var:maa_manager_selected_target_scope
			RECEIVER = root.var:maa_manager_selected_regiment_scope
		}
		if = {
			limit = { NOT = { exists = var:maa_manager_selected_target_scope } }
			var:maa_manager_selected_regiment_scope = { save_scope_as = regiment }
			maam_board_select_effect = yes
		}
	}
}

# Signed soldier plan in maa_total_planned_transfers:
#   > 0 selected -> partner, < 0 partner -> selected. Step = one stack's soldiers.
maam_board_soldier_limits = {
	save_temporary_scope_value_as = {
		name = maam_sol_max
		value = {
			value = var:maa_manager_selected_regiment_scope.maa_current_troops_count
			max = {
				value = var:maa_manager_selected_target_scope.maa_max_troops_count
				subtract = var:maa_manager_selected_target_scope.maa_current_troops_count
			}
			min = 0
		}
	}
	save_temporary_scope_value_as = {
		name = maam_sol_min
		value = {
			value = var:maa_manager_selected_target_scope.maa_current_troops_count
			max = {
				value = var:maa_manager_selected_regiment_scope.maa_max_troops_count
				subtract = var:maa_manager_selected_regiment_scope.maa_current_troops_count
			}
			min = 0
			multiply = -1
		}
	}
	save_temporary_scope_value_as = {
		name = maam_sol_step
		value = {
			value = var:maa_manager_selected_regiment_scope.maa_max_troops_count
			divide = var:maa_manager_selected_regiment_scope.maa_size
		}
	}
}

maam_board_soldier_step_effect = {
	if = {
		limit = { maam_board_actions_allowed = yes maa_manager_transfer_selection_valid = yes }
		maam_board_soldier_limits = yes
		if = {
			limit = { NOT = { has_variable = maa_total_planned_transfers } }
			set_variable = { name = maa_total_planned_transfers value = 0 }
		}
		change_variable = { name = maa_total_planned_transfers add = { value = scope:maam_sol_step multiply = $DIR$ } }
		set_variable = {
			name = maa_total_planned_transfers
			value = { value = var:maa_total_planned_transfers max = scope:maam_sol_max min = scope:maam_sol_min }
		}
		remove_variable = maam_last_from
		remove_variable = maam_last_to
	}
}

maam_board_soldier_fill_partner_effect = {
	if = {
		limit = { maam_board_actions_allowed = yes maa_manager_transfer_selection_valid = yes }
		maam_board_soldier_limits = yes
		set_variable = { name = maa_total_planned_transfers value = scope:maam_sol_max }
		remove_variable = maam_last_from
		remove_variable = maam_last_to
	}
}

maam_board_soldier_fill_selected_effect = {
	if = {
		limit = { maam_board_actions_allowed = yes maa_manager_transfer_selection_valid = yes }
		maam_board_soldier_limits = yes
		set_variable = { name = maa_total_planned_transfers value = scope:maam_sol_min }
		remove_variable = maam_last_from
		remove_variable = maam_last_to
	}
}

# Apply the signed plan, bounded again by the current numbers; keep the pair.
maam_board_send_soldiers_effect = {
	if = {
		limit = {
			maam_board_actions_allowed = yes
			maa_manager_transfer_selection_valid = yes
			has_variable = maa_total_planned_transfers
			NOT = { var:maa_total_planned_transfers = 0 }
		}
		maam_board_soldier_limits = yes
		save_temporary_scope_value_as = {
			name = maam_send_amount
			value = { value = var:maa_total_planned_transfers max = scope:maam_sol_max min = scope:maam_sol_min }
		}
		if = {
			limit = { scope:maam_send_amount > 0 }
			var:maa_manager_selected_regiment_scope = { change_maa_troops_count = { value = scope:maam_send_amount multiply = -1 } }
			var:maa_manager_selected_target_scope = { change_maa_troops_count = scope:maam_send_amount }
		}
		else_if = {
			limit = { scope:maam_send_amount < 0 }
			var:maa_manager_selected_target_scope = { change_maa_troops_count = scope:maam_send_amount }
			var:maa_manager_selected_regiment_scope = { change_maa_troops_count = { value = scope:maam_send_amount multiply = -1 } }
		}
		remove_variable = maam_last_from
		remove_variable = maam_last_to
		maam_board_reset_plan_effect = yes
	}
}

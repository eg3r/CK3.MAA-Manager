######################################################
# MAA Manager - Retrain armory: currency pieces for prices and fees.
# Each piece is one currency (money = gold or treasury, as the purchase took it; barter goods, prestige,
# piety, herd) and is empty when that amount is 0, so a line shows only what
# is involved: [GetPlayer.Custom('maam_rt_cost_fee_m')][...fee_ba')]...
# Pieces end with a space. ROOT = the player.
######################################################

maam_rt_cost_src_m = {
	type = character
	text = {
		trigger = { exists = var:maam_rt_src_price var:maam_rt_src_price > 0 has_variable = maam_rt_money_treasury }
		localization_key = MAAM_RT_COST_SRC_T
	}
	text = {
		trigger = { exists = var:maam_rt_src_price var:maam_rt_src_price > 0 }
		localization_key = MAAM_RT_COST_SRC_G
	}
	text = {
		localization_key = MAAM_EMPTY
		fallback = yes
	}
}
